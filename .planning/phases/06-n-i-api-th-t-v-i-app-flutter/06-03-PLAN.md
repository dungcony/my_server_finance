    // 2. Chuyển 500.000đ giữa 2 ví.
    await apiClient.post('/wallets/transfer', body: {
      'wallet_id': walletA,
      'destination_wallet_id': walletB,
      'amount': 500000,
      'date': DateTime.now().toIso8601String().substring(0, 10),
    });

    // 3. Tổng chi tháng phải GIỮ NGUYÊN.
    final homeAfter = await apiClient.get('/reports/home');
    final totalExpenseAfter = homeAfter['total_expense'] ?? homeAfter['summary']?['total_expense'];

    expect(totalExpenseAfter, totalExpenseBefore);

    await apiClient.clearSession();
  });
}

import 'package:finance_ai/core/network/api_client.dart';
import 'package:finance_ai/core/network/token_storage.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';

/// Phép thử §2.3 CHUYEN-SANG-API-THAT.md: sửa chi 100k ví A -> 80k ví B thì
/// ví A hoàn đủ 100k, ví B trừ 80k (không phải chỉ trừ chênh lệch 20k).
void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('SC2c: sửa giao dịch hoàn tác đúng 3 bước', (tester) async {
    final tokenStorage = TokenStorage();
    final apiClient = ApiClient(tokenStorage: tokenStorage, onSessionExpired: () {});

    final loginResponse = await apiClient.post('/auth/login', body: {
      'email': 'seed.user@datn.local',
      'password': 'SeedPass123!',
    });
    await tokenStorage.save(AuthTokens(
      accessToken: loginResponse['access_token'] as String,
      refreshToken: loginResponse['refresh_token'] as String,
    ));

    final wallets = await apiClient.getList('/wallets');
    final walletA = (wallets[0] as Map<String, dynamic>)['id'] as String;
    final walletB = (wallets[1] as Map<String, dynamic>)['id'] as String;

    final walletADetailBefore = await apiClient.get('/wallets/$walletA');
    final balanceABefore = walletADetailBefore['current_balance'] as int;
    final walletBDetailBefore = await apiClient.get('/wallets/$walletB');
    final balanceBBefore = walletBDetailBefore['current_balance'] as int;

    // 1. Ghi chi 100.000đ ở ví A.
    final categories = await apiClient.getList('/categories');
    final expenseCategory = categories.cast<Map<String, dynamic>>().firstWhere(
        (c) => c['type'] == 'expense' && c['parent_category_id'] == null);
    final createResponse = await apiClient.post('/transactions', body: {
      'type': 'expense',
      'amount': 100000,
      'wallet_id': walletA,
      'category_id': expenseCategory['id'],
      'date': DateTime.now().toIso8601String().substring(0, 10),
    });
    final txnId = (createResponse['transaction'] as Map<String, dynamic>)['id'];

    // 2. Sửa thành chi 80.000đ ở ví B.
    await apiClient.put('/transactions/$txnId', body: {
      'type': 'expense',
      'amount': 80000,
      'wallet_id': walletB,
      'category_id': expenseCategory['id'],
      'date': DateTime.now().toIso8601String().substring(0, 10),
    });

    // 3. Ví A phải HOÀN ĐỦ 100.000đ (số dư quay lại đúng bằng trước khi ghi giao dịch gốc).
    final walletADetailAfter = await apiClient.get('/wallets/$walletA');
    final balanceAAfter = walletADetailAfter['current_balance'] as int;
    expect(balanceAAfter, balanceABefore);

    // 4. Ví B chỉ trừ đúng 80.000đ — KHÔNG được trừ 20.000đ (sai kinh điển: chỉ trừ chênh lệch).
    final walletBDetailAfter = await apiClient.get('/wallets/$walletB');
    final balanceBAfter = walletBDetailAfter['current_balance'] as int;
    expect(balanceBAfter, balanceBBefore - 80000);

    await apiClient.clearSession();
  });
}

import 'package:finance_ai/core/network/api_client.dart';
import 'package:finance_ai/core/network/token_storage.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';

/// SC4 / D-13a: gửi 2 POST /transactions cùng Idempotency-Key chỉ sinh 1 giao dịch.
/// Idempotency-Key CỐ ĐỊNH (không dùng UUID ngẫu nhiên như luồng UI bình thường) —
/// ApiClient.post đã hỗ trợ sẵn tham số idempotencyKey (api_client.dart dòng 90-103).
void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('SC4: Idempotency-Key trùng chỉ sinh 1 giao dịch', (tester) async {
    final tokenStorage = TokenStorage();
    final apiClient = ApiClient(tokenStorage: tokenStorage, onSessionExpired: () {});

    final loginResponse = await apiClient.post('/auth/login', body: {
      'email': 'seed.user@datn.local',
      'password': 'SeedPass123!',
    });
    await tokenStorage.save(AuthTokens(
      accessToken: loginResponse['access_token'] as String,
      refreshToken: loginResponse['refresh_token'] as String,
    ));

    final wallets = await apiClient.getList('/wallets');
    final walletId = (wallets.first as Map<String, dynamic>)['id'] as String;
    final categories = await apiClient.getList('/categories');
    final expenseCategory = categories.cast<Map<String, dynamic>>().firstWhere(
        (c) => c['type'] == 'expense' && c['parent_category_id'] == null);

    const fixedKey = 'integration-test-fixed-key-001';
    final payload = {
      'type': 'expense',
      'amount': 15000,
      'wallet_id': walletId,
      'category_id': expenseCategory['id'],
      'date': DateTime.now().toIso8601String().substring(0, 10),
    };

    // 1. Gửi lần đầu — 201, ghi lại id giao dịch.
    final r1 = await apiClient.post('/transactions', body: payload, idempotencyKey: fixedKey);
    final id1 = (r1['transaction'] as Map<String, dynamic>)['id'];

    // 2. Gửi lại Y HỆT với cùng key — phải nhận LẠI cùng id, không tạo bản ghi mới.
    final r2 = await apiClient.post('/transactions', body: payload, idempotencyKey: fixedKey);
    final id2 = (r2['transaction'] as Map<String, dynamic>)['id'];

    expect(id2, id1);

    // 3. Đối chiếu bằng cách đếm số giao dịch amount=15000 tạo trong hôm nay qua GET /transactions —
    // giới hạn: đây là đếm qua API, không phải truy vấn CSDL trực tiếp (D-14 khuyến nghị SQL trực
    // tiếp cho môi trường có quyền truy cập JDBC; app không có, dùng API là phương án khả thi nhất
    // từ phía Flutter). Kèm ghi chú: nên bổ sung kiểm tra CSDL thủ công (psql) khi review D-14.
    final list = await apiClient.getPaged('/transactions', (json) => json,
        query: {'from_date': DateTime.now().toIso8601String().substring(0, 10)});
    final matchingCount = list.items
        .cast<Map<String, dynamic>>()
        .where((t) => t['amount'] == 15000 && t['id'] == id1)
        .length;
    expect(matchingCount, 1);

    await apiClient.clearSession();
  });
}
