```dart
import 'package:finance_ai/core/network/api_client.dart';
import 'package:finance_ai/core/network/token_storage.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';

/// SC5 / D-13b: nhiều request cùng hết hạn thẻ chỉ gây MỘT lần gọi /auth/refresh,
/// phiên không bị thu hồi oan. Dùng TokenStorage.saveForTest (test-only, xem token_storage.dart)
/// để seed access token đã hết hạn mà không cần chờ 3600s thật.
///
/// GIỚI HẠN: không đo trực tiếp số lần /auth/refresh được gọi từ phía app test (D-14 khuyến nghị
/// đếm qua log backend/bảng refresh_tokens) — test này chỉ xác nhận HÀNH VI BÊN NGOÀI đúng: cả 3
/// request song song đều cuối cùng thành công (200), và phiên vẫn dùng được sau đó (không bị thu
/// hồi oan). Đếm số lần /auth/refresh thật cần thử tay + xem log backend, xem D-14/D-13.
void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('SC5: nhiều request cùng hết hạn thẻ chỉ gộp 1 lần refresh', (tester) async {
    final tokenStorage = TokenStorage();
    final apiClient = ApiClient(tokenStorage: tokenStorage, onSessionExpired: () {});

    // 1. Đăng nhập thật để có refresh_token hợp lệ.
    final loginResponse = await apiClient.post('/auth/login', body: {
      'email': 'seed.user@datn.local',
      'password': 'SeedPass123!',
    });
    final validRefreshToken = loginResponse['refresh_token'] as String;

    // 2. Ghi đè bằng access token GIẢ HẾT HẠN (chuỗi JWT hợp lệ về hình thức nhưng exp quá khứ),
    // giữ nguyên refresh_token THẬT — backend chỉ cần refresh_token hợp lệ để cấp thẻ mới.
    const expiredAccessToken =
        'eyJhbGciOiJIUzI1NiJ9.eyJzdWIiOiJleHBpcmVkIiwiZXhwIjoxfQ.invalid-signature-force-expired-path';
    await tokenStorage.saveForTest(
      AuthTokens(accessToken: expiredAccessToken, refreshToken: validRefreshToken),
    );

    // 3. Bắn 3 request song song KHÔNG await tuần tự — mô phỏng nhiều màn hình cùng gọi lúc thẻ hết hạn.
    final results = await Future.wait([
      apiClient.getList('/wallets'),
      apiClient.getList('/categories'),
      apiClient.getPaged('/transactions', (json) => json),
    ]);

    // 4. Cả 3 đều phải thành công sau khi AuthInterceptor tự làm mới thẻ ngầm.
    expect(results[0], isNotNull);
    expect(results[1], isNotNull);
    expect(results[2], isNotNull);

    // 5. Phiên KHÔNG bị thu hồi oan — gọi thêm 1 request nữa vẫn phải thành công.
    final me = await apiClient.get('/auth/me');
    expect(me['email'], 'seed.user@datn.local');

    await apiClient.clearSession();
  });
}
```

```dart
import 'package:finance_ai/core/network/api_client.dart';
import 'package:finance_ai/core/network/token_storage.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';

/// Bước 6 D-18: kiểm nối nhóm budgets/reports trên backend thật. D-11 (06-CONTEXT.md) từng nghi
/// ngờ /budgets/suggestion và /reports/by-category-group có thể thiếu — RESEARCH.md đã xác minh
/// cả hai TỒN TẠI (BudgetController.java:61, ReportController.java:64). Test này chỉ xác nhận app

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('Budgets/Reports: 4 endpoint gọi thông trên backend thật', (tester) async {
    final tokenStorage = TokenStorage();
    final apiClient = ApiClient(tokenStorage: tokenStorage, onSessionExpired: () {});

    final loginResponse = await apiClient.post('/auth/login', body: {
      'email': 'seed.user@datn.local',
      'password': 'SeedPass123!',
    });
    await tokenStorage.save(AuthTokens(
      accessToken: loginResponse['access_token'] as String,
      refreshToken: loginResponse['refresh_token'] as String,
    ));

    final budgets = await apiClient.getList('/budgets');
    expect(budgets, isNotEmpty, reason: 'DevDataSeeder (06-01) đã seed 1 ngân sách cho user thử');

    final suggestion = await apiClient.get('/budgets/suggestion', query: {
      'category_id': (budgets.first as Map<String, dynamic>)['category_id'],
    });
    expect(suggestion, isNotNull);

    final byCategoryGroup = await apiClient.getList('/reports/by-category-group');
    expect(byCategoryGroup, isA<List<dynamic>>());

    final dailyTrend = await apiClient.get('/reports/daily-trend');
    expect(dailyTrend, isNotNull);

    await apiClient.clearSession();
  });
}
```
