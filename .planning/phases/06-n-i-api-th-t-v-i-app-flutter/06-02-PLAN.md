import '../../config/app_config.dart';
import 'mock_router.dart';

/// Chặn mọi yêu cầu và trả dữ liệu mẫu có trạng thái, dùng khi
/// `USE_MOCK=true`, HOẶC luôn chặn riêng nhóm `/ai/*` bất kể cờ (D-02,
/// 06-CONTEXT.md) — backend Phase 5 (AiController) chưa tồn tại. Gỡ ngoại lệ
/// này khi Phase 5 xong.
///
/// Đặt ở tầng Dio chứ không phải tầng repository là có chủ đích: repository,
/// provider và giao diện chạy **đúng một đường dẫn mã** dù dữ liệu đến từ
/// đâu. Bật tắt phần còn lại bằng `AppConfig.useMock`.
class MockInterceptor extends Interceptor {
  /// D-02: nhóm `ai` luôn dùng mock kể cả khi USE_MOCK=false.
  static const _alwaysMockPrefixes = ['/ai/'];

  @override
  Future<void> onRequest(
    RequestOptions options,
    RequestInterceptorHandler handler,
  ) async {
    final forceMock = _alwaysMockPrefixes.any(options.path.startsWith);
    if (!AppConfig.useMock && !forceMock) {
      handler.next(options);
      return;
    }
    final result = await MockRouter.instance.handle(options);
    handler.resolve(_response(options, result.status, result.json));
  }

  Response<dynamic> _response(
    RequestOptions options,
    int status,
    Map<String, dynamic> json,
  ) => Response<dynamic>(
    requestOptions: options,
    statusCode: status,
    data: json,
  );
}
```

import 'package:finance_ai/core/network/api_client.dart';
import 'package:finance_ai/core/network/token_storage.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';

/// Chạy với: flutter test integration_test/real_backend/category_rollup_test.dart
///   --dart-define=USE_MOCK=false --dart-define=API_BASE_URL=http://10.0.2.2:8080/v1 -d <device>
/// Phép thử §2.1 CHUYEN-SANG-API-THAT.md: chi vào con "Cà phê" phải hiện khi lọc theo cha "Ăn uống".
void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('SC2a: cộng gộp danh mục con vào cha', (tester) async {
    final tokenStorage = TokenStorage();
    final apiClient = ApiClient(tokenStorage: tokenStorage, onSessionExpired: () {});

    final loginResponse = await apiClient.post('/auth/login', body: {
      'email': 'seed.user@datn.local',
      'password': 'SeedPass123!',
    });
    await tokenStorage.save(AuthTokens(
      accessToken: loginResponse['access_token'] as String,
      refreshToken: loginResponse['refresh_token'] as String,
    ));

    // 1. Tìm id danh mục cha "Ăn uống" và con "Cà phê" qua GET /categories.
    final categories = await apiClient.getList('/categories');
    final anUong = categories.cast<Map<String, dynamic>>().firstWhere(
        (c) => c['name'] == 'Ăn uống' && c['parent_category_id'] == null);
    final children = (anUong['children'] as List<dynamic>?) ?? const [];
    final caPhe = children.cast<Map<String, dynamic>>().firstWhere((c) => c['name'] == 'Cà phê');

    // 2. Lấy 1 ví để ghi giao dịch.
    final wallets = await apiClient.getList('/wallets');
    final walletId = (wallets.first as Map<String, dynamic>)['id'] as String;

    // 3. Ghi khoản chi 25.000đ vào danh mục con "Cà phê".
    final txnResponse = await apiClient.post('/transactions', body: {
      'type': 'expense',
      'amount': 25000,
      'wallet_id': walletId,
      'category_id': caPhe['id'],
      'date': DateTime.now().toIso8601String().substring(0, 10),
    });
    final createdId = (txnResponse['transaction'] as Map<String, dynamic>)['id'];

    // 4. Lọc theo danh mục CHA "Ăn uống" — giao dịch phải xuất hiện dù gán vào con.
    final filtered = await apiClient.getPaged(
      '/transactions',
      (json) => json,
      query: {'category_id': anUong['id']},
    );
    final ids = filtered.items.map((e) => (e as Map<String, dynamic>)['id']);
    expect(ids, contains(createdId));

    await apiClient.clearSession();
  });
}
```
