    @Slf4j
    public class BudgetAlertListener {
        private final BudgetProgressRepository budgetProgressRepository;
        private final NotificationRepository notificationRepository;

        @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
        public void onTransactionRecorded(TransactionRecordedEvent event) {
            if (!"expense".equals(event.type()) || event.categoryId() == null) return;
            try {
                List<BudgetProgressProjection> affected = budgetProgressRepository
                        .findAllForUser(event.userId(), true, null).stream()
                        .filter(bp -> bp.getCategoryId().equals(event.categoryId())
                                || /* cộng gộp: ngân sách đặt ở danh mục CHA của event.categoryId() */)
                        .toList();
                for (var bp : affected) {
                    if ("normal".equals(bp.getStatus())) continue;
                    String severity = "over_limit".equals(bp.getStatus()) ? "critical" : "alert";
